\newglossaryentry{maninthemiddle}{
  name=Man In The Middle,
  description={Ataque à redes de computadores no qual uma pessoa não
  autorizada intercepta indevidamente o tráfego de informações},
}

\newglossaryentry{checksum}{
  name=Checksum,
  description={Soma de verificação dos dados para análise de erros,
    em casos onde houver alterações durante o armazenamento ou
  transmissão é solicitado uma retransmissão do pacote},
}

\newglossaryentry{opensource}{
  name=Open Source,
  description={Modelo de desenvolvimento de software onde o
    código-fonte é disponibilizado publicamente para que qualquer
  pessoa possa ver, modificar, melhorar e distribuir},
}
